ALGORITHM SearchBooks
READ query text and category
SET matches to empty list
FOR EACH book in catalogue
    IF title contains query AND category matches (or category is All)
        ADD book to matches
    END IF
END FOR
SET page = 0
DISPLAY current page
END ALGORITHM

ALGORITHM DisplayPage
CLEAR result table
CALCULATE start = page * pageSize
CALCULATE end = minimum(start + pageSize, number of matches)
DISPLAY records from start to end
UPDATE status message
END ALGORITHM
