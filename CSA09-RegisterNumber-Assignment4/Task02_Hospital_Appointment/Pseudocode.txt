ALGORITHM RefreshDoctorList
READ selected department
CLEAR doctor combo box
FOR EACH doctor in doctorMap[selected department]
    ADD doctor to combo box
END FOR
END ALGORITHM

ALGORITHM BookAppointment
SET bookingInProgress = true
READ patient ID, department, selected doctor and time slot
IF any required value is missing THEN display validation error
ELSE IF doctor/time slot is already booked THEN display conflict
ELSE save booking and display confirmation
END IF
SET bookingInProgress = false
END ALGORITHM
